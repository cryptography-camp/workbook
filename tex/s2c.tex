\section{Sign-to-Contract}

\subsection{Syntax \& Correctness}

\begin{definition}[Sign-to-Contract Signature Scheme]
  A sign-to-contract-compatible signature scheme $\algo{SigS2C}$ is a tuple of polynomial-time algorithms $(\algo{Setup}, \algo{KeyGen}, \algo{Sign}, \algo{Verify}, \algo{VerifyCom})$ where:
  \begin{itemize}
    \item $\algo{Setup}(1^\secpar) \rightarrow \params$ is a probabilistic algorithm that takes the security parameter as input and outputs public parameters $\params$ including the message space $\mathcal{M}$, the committed message space $\mathcal{MC}$, the signature space $\mathcal{S}$, and randomness space $\mathcal{R}$.
    \item $\algo{KeyGen}(\params) \rightarrow (\sk, \pk)$ is a probabilistic algorithm that takes public parameters $\params$ as input and outputs a secret key $\sk$ and a public key $\pk$.
    \item $\algo{Gen}(\params) \rightarrow (r_C)$: is a probabilistic algorithm that takes public parameters \params as input and outputs randomness $r_C$.
    \item $\algo{Sign}(\params, \sk, m, m_C, r_C) \rightarrow (\sigma, R_C)$ is a deterministic algorithm that takes public parameters $\params$, a secret key $\sk$, message $m \in \mathcal{M}$, a message $m_C \in \mathcal{MC}$ and randomness $r_C$ as input and outputs a signature $\sigma \in \mathcal{S}$ that commits to $m_C$ and randomness $R_C$.
    \item $\algo{Verify}(\params, \pk, m, \sigma) \rightarrow \{0, 1\}$ is a deterministic algorithm that takes public parameters $\params$, a public key $\pk$, a message $m \in \mathcal{M}$, and a signature $\sigma \in \mathcal{S}$ as input and outputs $1$ (accept) if the signature is valid, and outputs $0$ (reject) otherwise.
    \item $\algo{VerifyCom}(\params, \sigma, m_C, R_C) \rightarrow \{0, 1\}$ is a deterministic algorithm that takes public parameters $\params$, a signature $\sigma \in \mathcal{S}$, a message $m_C \in \mathcal{MC}$ and randomness $R_C$ as input and outputs $1$ (accept) if the commitment commits to $m_C$ with randomness $R_C$, and outputs $0$ (reject) otherwise.
  \end{itemize}
\end{definition}

\begin{definition}[Correctness]\label{def:s2c-correctness}
  A sign-to-contract-compatible signature scheme $\algo{SigS2C} = (\algo{Setup}, \algo{KeyGen}, \algo{Sign}, \algo{Verify}, \algo{VerifyCom})$ is \emph{correct} if for all $\secpar \in \NN$, all $\params \in \algo{Setup}(1^\secpar)$, all $(\sk, \pk) \in \algo{KeyGen}(\params)$, all $r_C \in \algo{Gen}(\params)$, all messages $m \in \mathcal{M}$ and $m_C \in \mathcal{MC}$:
  \[
    \Pr\left[
    \begin{array}{c}
      \algo{Verify}(\params, \pk, m, \sigma) = 1 \\
      \wedge \\
      \algo{VerifyCom}(\params, \sigma, m_C, R_C) = 1
    \end{array}
    \middle|
    \begin{array}{c}
      (\sigma, R_C) \gets \\
      \algo{Sign}(\params, \sk, m, m_C, r_C)
    \end{array}
    \right] = 1
  \]
  where the probability is taken over the randomness of $\algo{Sign}$.
\end{definition}

\subsection{Security}

\begin{definition}[EUF-CC-CMA Security for Sign-to-Contract]\label{def:euf-cma-s2c}
  A sign-to-contract signature scheme $\algo{SigS2C}$ is \emph{existentially unforgeable under chosen-commitment and chosen-message attacks (EUF-CC-CMA)} if for all \ppt adversaries $\adv$:
  \[
    \advantage{EUF-CC-CMA}{\adv, \algo{SigS2C}} \defeq \pr{\game{\Game~\algo{EUF-CC-CMA}}{\algo{SigS2C}} = 1} = \negl
  \]
  where $\game{\Game~\algo{EUF-CC-CMA}}{\algo{SigS2C}}$ is defined in \autoref{fig:euf-cma-s2c}.
\end{definition}

	A sign-to-contract signature scheme $\algo{SigS2C}$ is \emph{strongly binding} if for all \ppt adversaries $\adv$:
	\[
	  \advantage{ComStrongBind}{\adv, \algo{SigS2C}} \defeq \pr{\game{\Game~\algo{ComStrongBind }}{\algo{SigS2C}} = 1} = \negl

\subsection{Schnorr Signature with Sign-to-Contract}

\begin{definition}[Schnorr Signature Scheme with Sign-to-Contract]\label{def:schnorr-s2c}
  Let $\grgen$ be a group generation algorithm, $\algo{H}_{\mathsf{sig}}$ be a hash function with output space $\ZZ_p$ for signatures, and $\algo{H}_{\mathsf{com}}$ be a hash function with output space $\ZZ_p$ for commitments. The Schnorr sign-to-contract signature scheme $\algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]$ is defined as follows:
  \begin{itemize}
    \item $\algo{Setup}(1^\secpar) \rightarrow \params$: Run $\gparam \gets \grgen(1^\secpar)$, $\kappa_{\mathsf{sig}} \gets \algo{H}_{\mathsf{sig}}.\algo{Gen}(1^\secpar)$, and $\kappa_{\mathsf{com}} \gets \algo{H}_{\mathsf{com}}.\algo{Gen}(1^\secpar)$. The message and committed message spaces are $\mathcal{M} = \{0,1\}^*$ and $\mathcal{MC} = \{0,1\}^*$, and the signature space is $\mathcal{S} = \GG \times \ZZ_p$. Output $\params = (\gparam, \kappa_{\mathsf{sig}}, \kappa_{\mathsf{com}}, \mathcal{M}, \mathcal{MC}, \mathcal{S})$.
    
    \item $\algo{KeyGen}(\params) \rightarrow (\sk, \pk)$: Sample $x \sample \ZZ_p$. Compute $X = g^x$. Output $\sk = x$ and $\pk = X$.
    
    \item $\algo{Gen}(\params) \rightarrow r_C$: Sample $x \sample \ZZ_p$. Output $r_C = x$.

    \item $\algo{Sign}(\params, \sk, m, m_C, r_C) \rightarrow (\sigma, R_C)$: Parse $\params = (\gparam, \kappa_{\mathsf{sig}}, \kappa_{\mathsf{com}}, \mathcal{M}, \mathcal{MC}, \mathcal{S})$ and $\sk = x$. Compute:
    \begin{align*}
      R_C &= g^{r_C} \\
      r &= r_C + \algo{H}_{\mathsf{com}}.\algo{Eval}(\kappa_{\mathsf{com}}, (R_C, m_C))  \bmod p\\
      R &= g^r \\
      c &= \algo{H}_{\mathsf{sig}}.\algo{Eval}(\kappa_{\mathsf{sig}}, (R, m)) \\
      s &= r + c \cdot x \bmod p
    \end{align*}
    Output $\sigma = (R, s)$ and $R_C$.
    
    % TODO: Is this missing a (mod p) on the result of H_sig like we do below for H_com, since it returns elements of \ZZ_p?
    \item $\algo{Verify}(\params, \pk, m, \sigma) \rightarrow \{0, 1\}$: Parse $\params = (\gparam, \kappa_{\mathsf{sig}}, \kappa_{\mathsf{com}}, \mathcal{M}, \mathcal{MC}, \mathcal{S})$, $\pk = X$ and $\sigma = (R, s)$. Compute $c = \algo{H}_{\mathsf{sig}}.\algo{Eval}(\kappa_{\mathsf{sig}}, (R, m))$. Accept if and only if:
    \[
      g^s = R \cdot X^c
    \]

    \item $\algo{VerifyCom}(\params, \sigma, m_c, R_C) \rightarrow \{0, 1\}$: Parse $\params = (\gparam, \kappa_{\mathsf{sig}}, \kappa_{\mathsf{com}}, \mathcal{M}, \mathcal{MC}, \mathcal{S})$, $\sigma = (R, s)$. Compute $c = \algo{H}_{\mathsf{com}}.\algo{Eval}(\kappa_{\mathsf{com}}, (R_C, m_C)) \bmod p$. Accept if and only if:
    \[
      R = R_C \cdot g^c
    \]
  \end{itemize}
\end{definition}

\begin{lemma}[Correctness of Schnorr Signatures with Sign-to-Contract]\label{lem:schnorr-s2c-correctness}
  The Schnorr signature with sign-to-contract scheme\\ $\algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]$ is correct.
\end{lemma}

\begin{proof}
  Let $(\sigma = (R, s), R_C)$ be the output of $\algo{Sign}(\params, \sk, m, m_c)$ where $\sk = x$ and $\pk = X = g^x$. By the signing algorithm, we have $R_C = g^{r_c}$ for some $r_c \sample \ZZ_p$, $c_c = \algo{H}_{\mathsf{com}}.\algo{Eval}(\kappa_{\mathsf{com}}, (R_C, m_c))$, $R = g^r$ for $r = r_C + c_c$, $c_s = \algo{H}_{\mathsf{sig}}.\algo{Eval}(\kappa_{\mathsf{sig}}, (R, m))$, and $s = r + c_s \cdot x \bmod p$. The verification equation holds because:
  \[
    g^s = g^{r + c_s \cdot x} = g^r \cdot g^{c_s \cdot x} = R \cdot (g^x)^{c_s} = R \cdot X^{c_s}
  \]
  Furthermore, the commitment opening equation holds because:
  \[
    R = g^r = g^{r_c + c_c} = R_C \cdot g^{c_c}
  \]
  Therefore, $\algo{Verify}(\params, \pk, m, \sigma) = 1 \wedge \algo{VerifyCom}(\params, \sigma, m_C, R_C) = 1$ for all $(\sigma, R_C)$ generated by $\algo{Sign}(\params, \sk, m, m_C)$.
\end{proof}


\begin{theorem}[Schnorr signatures with sign-to-contract are EUF-CC-CMA secure]\label{thm:schnorr-s2c-euf-cma}
  Let $\grgen$ be a group generation algorithm for which the discrete logarithm problem is hard and let $\algo{H}_{\mathsf{sig}}$ and $\algo{H}_{\mathsf{com}}$ be hash functions with output space $\ZZ_p$.
  Then the Schnorr signature with sign-to-contract scheme $\algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]$ is EUF-CC-CMA secure in the random oracle model for $\algo{H}_{\mathsf{sig}}$ and $\algo{H}_{\mathsf{com}}$.
  More precisely, for any \ppt adversary $\adv$ against the EUF-CC-CMA security of $\algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]$ making at most $q_s$ queries to the signing oracle, at most $q_{h_s}$ queries to the random oracle $\algo{H}_{\mathsf{sig}}$ and at most $q_{h_c}$ queries to the random oracle $\algo{H}_{\mathsf{com}}$,
  there exists a \ppt adversary $\bdv$ against the EUF-CMA security of $\algo{SchnorrSig}[\grgen, \algo{H}_{\mathsf{sig}}]$ making at most $q_s$ queries to the signing oracle and at most $q_{h_s}$ queries to the random oracle $\algo{H}_{\mathsf{sig}}$ such that
  \[
    \advantage{EUF-CC-CMA}{\adv, \algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]} \leq \advantage{EUF-CMA}{\bdv, \algo{SchnorrSig}[\grgen, \algo{H}_{\mathsf{sig}}]} + q_s \frac{q_s + q_{h_C}}{2^{\secpar - 1}}.
  \]
\end{theorem}


\begin{proof}
  Let $\adv$ be a \ppt adversary against the EUF-CC-CMA security of $\algo{SchnorrS2C}[\grgen, \algo{H}_{\mathsf{sig}}, \algo{H}_{\mathsf{com}}]$  making at most $q_s$ queries to the signing oracle, at most $q_{h_s}$ queries to the random oracle $\algo{H}_{\mathsf{sig}}$ and at most $q_{h_c}$ queries to the random oracle $\algo{H}_{\mathsf{com}}$.

  \begin{enumerate}
  \item Let $\bdv$ be the adversary as defined in \autoref{fig:schnorr-s2c-reduction} that internall runs $\adv$.
  \item Let $E$ be the event that during a $\pcoracle{S2CSign}$ query, $\algo{H}_{\mathsf{com}}$ had already been queried with $R_C, m_C$.
  \item Claim: Then \[
      \pr{\game{\Game~\algo{EUF-CC-CMA}}{\algo{SchnorrS2C}} = 1 \wedge \neg E} =  \pr{\game{\Game~\algo{EUF-CMA}}{\algo{SchnorrSig}}[\bdv] = 1 \wedge \neg E}
  \]
  \begin{enumerate}
    \item In both games, the inputs of $\adv$ have the same distribution. The only change is in oracle $\pcoracle{S2CSign}$.
  \item In both games $\sigma$ is a valid Schnorr signature.
    In particular, in $\bdv$ it is just the output of $\pcoracle{Sign}$.
    % TODO: should it say "in S2CSign_B it is just the output of Sign" instead of "in B it is just the output of Sign"? It confused me for a second as i read it as talking about the output of B itself (which returns no the output of Sign but the output of A).
  \item In \bdv, we have $R_C = R g^{-c}$ where $c = \algo{H}_{\mathsf{com}}(R_C, m_C)$.
    Therefore, $R = R_C \cdot g^{c}$ for signature $(R, s) = \sigma$.
  \end{enumerate}
  \item $\pr{E} = q_s \frac{q_s + q_{h_C}}{2^{\secpar - 1}}$
    \begin{enumerate}
      \item Let $E_i$ be the event that the assertion is triggered in the $i-th$ call to $\pcoracle{SignS2C}$.
      \item At any call to $\pcoracle{SignS2C}$, the table $T$ has at most $q_s + q_{h_c}$ entries.
      \item Since $c$ is distributed uniformly at random in $\ZZ_p$, $R_C$ is distributed uniformly at random in $\GG$.
        If $E_i$ happens then there exists $R_C'$ in $T$ such that $R_C = R_C'$. Thus,
        \begin{align*}
          \pr{E_i} &\leq \frac{q_s + q_{h_C}}{|\GG|} \\
                   &\leq \frac{q_s + q_{h_C}}{2^{\secpar - 1}}
        \end{align*}
        where we used the fact that $|\GG| \ge 2^{\secpar - 1}$.
      \item We have
        \begin{align*}
          \pr{E} &= \pr{E_0 \vee \ldots \vee E_{q_s}}\\
           &\le \pr{E_0} + \ldots + \pr{E_{q_s}} \\
           &\le q_s \frac{q_s + q_{h_C}}{2^{\secpar - 1}} \\
        \end{align*}
    \end{enumerate}
  \item Using the difference lemma, we have
    \[
    | \pr{\game{\Game~\algo{EUF-CC-CMA}}{\algo{SchnorrS2C}} = 1} - \pr{\game{\Game~\algo{EUF-CMA}}{\algo{SchnorrSig}}[\bdv] = 1}| \le q_s \frac{q_s + q_{h_C}}{2^{\secpar - 1}}.
      \]
      Thus,
    \[
    \pr{\game{\Game~\algo{EUF-CC-CMA}}{\algo{SchnorrS2C}} = 1} \le \pr{\game{\Game~\algo{EUF-CMA}}{\algo{SchnorrSig}}[\bdv] = 1} + q_s \frac{q_s + q_{h_C}}{2^{\secpar - 1}}.
      \]
  \end{enumerate}
  It is easy to see that $\bdv$ is \ppt (if $\adv$ is \ppt).
  $\bdv$ makes no more than $q_s$ queries to $\pcoracle{Sign}$ and $q_{h_s}$ queries to $\algo{H}_{\mathsf{sig}}$.
\end{proof}
